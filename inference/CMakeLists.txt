cmake_minimum_required(VERSION 3.16)
project(h9dp_inference LANGUAGES CXX)

find_package(BZip2 QUIET)
find_package(ZLIB REQUIRED)

add_library(h9dp_inference inference_best.cpp hsd.cpp v1_features.cpp)
add_library(h9dp::inference ALIAS h9dp_inference)
target_compile_features(h9dp_inference PUBLIC cxx_std_20)
target_include_directories(h9dp_inference PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(h9dp_inference PRIVATE ZLIB::ZLIB)
if(BZip2_FOUND)
    target_link_libraries(h9dp_inference PRIVATE BZip2::BZip2)
    target_compile_definitions(h9dp_inference PRIVATE H9DP_HAVE_BZIP2)
else()
    message(STATUS "BZip2 not found; uncompressed DAT and gzip remain supported")
endif()
set_target_properties(h9dp_inference PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(h9dp_inference PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()

add_executable(inference_best_cli main.cpp)
target_link_libraries(inference_best_cli PRIVATE h9dp::inference)
